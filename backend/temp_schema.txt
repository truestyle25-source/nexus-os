// Decis+�o arquitetural (Fase 3, d+�vida t+�cnica registrada em
// DEBITO-TECNICO.md item 8): giro real precisa de estoque m+�dio no
// per+�odo, que por sua vez precisa de uma s+�rie hist+�rica de saldo di+�rio
// ��� a posi+�+�o atual (Estoque) e o log de eventos (EstoqueMovimentacao) n+�o
// bastam sozinhos porque reconstruir o saldo em cada dia passado a partir
// s+� do log seria caro de consultar toda vez. Uma linha por
// produto+�filial+�dia +� o suficiente (n+�o guarda hist+�rico de pre+�o por
// unidade nem lote ��� isso seria over-engineering para o que a Se+�+�o 9
// pede). Alimentada por src/scripts/gerarSnapshotsDiarios.ts (rodado uma
// vez por dia por um agendador externo ��� este projeto n+�o inclui um
// scheduler) ou por POST /estoque/snapshot/gerar.
// ===========================================================================
//  Ciclo 6 — Sessão de Inventário
// ===========================================================================
//
//  POR QUE UMA SESSÃO E NÃO UM CAMPO NA FOLHA
//  -----------------------------------------
//  A folha de inventário morre com a tela: fechar o navegador perde a contagem
//  inteira. Um inventário de depósito leva horas ou dias, e a empresa fecha a
//  loja às 18h com metade dela contada. A sessão é o registro que sobrevive ao
//  navegador — reabrir amanhã continua de onde parou.
//
//  O SALDO DO SISTEMA É CONGELADO NA ABERTURA
//  --------------------------------------------
//  `quantidadeSistema` é gravada no momento em que a sessão é criada e nunca
//  muda. Isso não é economia de consulta, é o que dá sentido ao número: se o
//  estoque muda no meio da contagem (uma venda, uma entrada de compra) e a
//  divergência fosse recalculada a cada leitura, o inventário compararia a
//  contagem física com um alvo que se moveu — e acharia desvio onde só houve
//  operação normal. Congelado, a divergência é a diferença entre o que estava
//  lá quando alguém abriu a folha e o que realmente existe.
//
//  A IDENTIFICAÇÃO DO ITEM É COPIADA, NÃO REFERENCIADA
//  ---------------------------------------------------
//  Produto, unidade e local são gravados como texto. Um inventário é documento
//  de trabalho: ele precisa continuar legível depois que o produto foi
//  renomeado ou desativado, e uma linha que vira "null" porque o cadastro
//  mudou é pior que inútil. O `estoqueId` fica só como rastro, e pode sumir.
//
//  VISÃO CEGA FICA REGISTRADA
//  --------------------------
//  `cegas` diz se aquela sessão foi contada sem o saldo visível. Um período
//  depois, o histórico mostra se a contagem foi cega — e o `criadoPorId` diz
//  quem abriu. É a diferença entre "inventário" e "conferência", e precisa ser
//  auditável.

enum StatusSessaoInventario {
  ABERTA
  CONCLUIDA
  CANCELADA
}

model SessaoInventario {
  id         String @id @default(cuid())
  empresaId  String
  /// Rótulo que a empresa dá à contagem ("Inventário 3T2026 — Centro").
  nome       String
  filialId   String
  filial     Filial @relation(fields: [filialId], references: [id])
  status     StatusSessaoInventario @default(ABERTA)
  /// `true` quando a folha foi aberta sem o saldo do sistema à vista.
  cegas      Boolean @default(true)
  /// Quem abriu a folha e quando — a autoria é o que dá valor ao histórico.
  criadoPorId String?
  criadoPor   Usuario? @relation("SessaoInventarioCriador", fields: [criadoPorId], references: [id])
  criadoEm   DateTime @default(now())
  /// Quem fechou a contagem. NULL enquanto a sessão está aberta.
  concluidoPorId String?
  concluidoPor   Usuario? @relation("SessaoInventarioConclusor", fields: [concluidoPorId], references: [id])
  concluidoEm DateTime?
  observacoes String?
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt

  contagens  ContagemInventario[]

  /// Duas folhas abertas na mesma filial transformam uma contagem em duas
  /// versões diferentes do mesmo número.
  @@index([empresaId, status])
  @@index([empresaId, filialId, criadoEm])
}

model ContagemInventario {
  id         String @id @default(cuid())
  empresaId  String
  sessaoId   String
  sessao     SessaoInventario @relation(fields: [sessaoId], references: [id], onDelete: Cascade)
  /// Rastro da linha de estoque contada. `SetNull`: o produto pode ser
  /// desativado ou a linha removida, e a contagem já feita continua valendo.
  estoqueId  String?
  estoque    Estoque? @relation(fields: [estoqueId], references: [id], onDelete: SetNull)

  // Cópia do que foi contado — o inventário precisa continuar legível.
  produtoId   String
  produtoNome String
  unidade     String?
  local       String?
  /// Saldo do sistema NO MOMENTO DA ABERTURA. Congelado, nunca recalculado.
  quantidadeSistema Decimal @db.Decimal(14, 3)
  /// NULL enquanto ninguém contou. É o que distingue "não contado" de
  /// "contado como zero" — os dois sãoZero no papel, e um deles é erro.
  quantidadeContada Decimal? @db.Decimal(14, 3)
  /// Quem contou este item e quando.
  contadoPorId String?
  contadoPor   Usuario? @relation("ContagemInventarioAutor", fields: [contadoPorId], references: [id])
  contadoEm   DateTime?
  observacoes String?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@unique([sessaoId, estoqueId])
  @@index([empresaId, sessaoId])
}

/// Documento anexado, guardado em bucket (S3 ou Supabase Storage).
///
/// POR QUE EXISTE
/// -------------
/// Antes as colunas `documentos` de Advertência/Suspensão/Desligamento eram
 /// `String[]` guardando apenas o NOME do arquivo — sem binário, sem caminho, sem
 /// chave. O nome aparecia na tela e no dossie, mas o arquivo não existia em lugar
 /// nenhum: era metadado de arquivo inexistente.
 ///
 /// Aqui fica o endereço real: `chave` é o caminho dentro do bucket e
 /// `bucket`/`provedor` dizem ONDE ele está. O download nunca usa `nomeArquivo`
 /// para montar caminho — usa `chave`, que é o que o provedor reconhece.
 ///
 /// SEGURANÇA
 /// ---------
 ///   • `empresaId` é obrigatório e toda leitura filtra por ele: documento de uma
 ///     empresa nunca é acessível por usuário de outra (isolamento por tenant).
 ///   • O bucket é PRIVADO. O acesso é por URL assinada com expiração curta
 ///     (`/rh/documentos/:id/download`), nunca por URL pública permanente.
 ///   • `contentType` e `tamanho` gravam o que foi declarado no upload e são
 ///     devolvidos no download — o `Content-Type` da resposta vem daqui, para o
 ///     navegador não abrir HTML onde se esperava PDF.
///
model Documento {
  id             String   @id @default(cuid())
  empresaId      String
  // Entidade dona do anexo: Funcionario | Advertencia | Suspensao | Desligamento
  // Guardada como string (e não como relação) porque o mesmo bucket serve a
  // várias telas; a validação de que a entidade existe e pertence à mesma
  // empresa acontece na rota de upload.
  entidadeTipo    String
  entidadeId      String
  /// Atalho de relacao quando o anexo e de um funcionário (dossiê). O
  /// `Documento` continua genérico: anexos de Advertência/Suspensão/Desligamento
  /// usam apenas `entidadeTipo`/`entidadeId`.
  funcionarioId   String?
  funcionario     Funcionario? @relation(fields: [funcionarioId], references: [id])
  // Nome original, só para exibição. NUNCA usado para montar caminho no bucket.
  nomeOriginal    String
  // Chave real dentro do bucket, ex.: "empresa-abc/rh/1699999999-contrato.pdf"
  chave           String   @unique
  bucket          String
  provedor        String   @default("s3") // s3 | supabase
  contentType     String
  tamanho         Int      // bytes
  // Descrição livre de onde veio (ex.: "Atestado de ponto").
  descricao       String?
  // Quem anexou — para auditoria de quem expõe documento de terceiro.
  enviadoPorId    String?
  enviadoPor      Usuario? @relation(fields: [enviadoPorId], references: [id])
  // "disponivel" | "arquivado" — o arquivamento NÃO apaga o objeto do bucket;
  // só o esconde do acesso operacional (ver janela de 30 dias no RH).
  situacao         String    @default("disponivel")
  comprovanteDe    PagamentoRH[]
  createdAt        DateTime  @default(now())

  @@index([empresaId, entidadeTipo, entidadeId])
  @@index([funcionarioId])
  @@index([empresaId, situacao])
  @@index([entidadeId])
}

// Desligamento / termo de rescisão (CLT art. 477 e seguintes).
//
// Não existia no schema: a aba "Desligamento" do RH montava o termo e as
// verbas apenas no estado do navegador, sem gravar nada, e ainda criava um
// registro DE EXEMPLO quando a lista estava vazia — o usuário via um termo
// com dados fictícios e, ao recarregar a tela, tudo sumia. Com este modelo o
// termo passa a ser um registro real, auditável e persistente.
model Desligamento {
  id              String      @id @default(cuid())
  empresaId       String
  funcionarioId   String
  funcionario     Funcionario @relation(fields: [funcionarioId], references: [id])
  data            DateTime    @db.Date
  // Motivo: sem-justa-causa | justa-causa | pedido | acordo | ato-improbidade
  tipo            String
  tipoLabel       String // rótulo exibido no termo (ex.: "Sem justa causa")
  art             String      // enquadramento legal (ex.: "ART. 487 CLT")
  motivo          String
  // aviso: trabalhado | indenizado | dispensado
  aviso           String      @default("indenizado")
  saldoDias       Int         @default(0)
  // Verbas calculadas no momento do registro e gravadas como SNAPSHOT.
  //
  // Guardadas separadamente porque o salário e as datas do cadastro podem
  // mudar depois, mas o termo já impresso precisa continuar fiel ao que foi
  // apurado no ato. A soma de `verbas13 + verbasFerias + verbasAviso +
  // verbasTotal` NÃO fecha sozinha: `verbasTotal` inclui também 1/3, férias
  // proporcionais e saldo de salário, que ganharam colunas próprias.
  verbas13        Decimal?    @db.Decimal(12, 2) // 13º proporcional
  verbasFerias    Decimal?    @db.Decimal(12, 2) // férias vencidas
  verbasAviso     Decimal?    @db.Decimal(12, 2) // aviso prévio indenizado
  verbasTotal     Decimal?    @db.Decimal(12, 2)
  // Detalhamento que o Art. 477 exige no termo. Antes estes valores eram
  // calculados no navegador com percentuais fixos e NÃO eram gravados: o PDF
  // saía com números que o banco não conseguia reproduzir.
  verbasTerco          Decimal? @db.Decimal(12, 2) // 1/3 constitucional
  verbasSaldo          Decimal? @db.Decimal(12, 2) // saldo de salário (em REAIS)
  verbasFeriasProporc  Decimal? @db.Decimal(12, 2) // férias proporcionais
  meses13              Int?     // meses considerados no 13º (informativo)
  anosCompletos        Int?     // anos completos de serviço (férias vencidas)
  diasAviso            Int?     // dias de aviso prévio gerados
  // FGTS e multa rescisória dependem do saldo DEPOSITADO do FGTS, que não
  // existe no schema (não há tabela de depósitos). Ficam nulos de propósito:
  // o termo os exibe como "não apurado" em vez de estimar por percentual.
  saldoFgts            Decimal? @db.Decimal(12, 2)
  multaFgts            Decimal? @db.Decimal(12, 2)
  fgtsApurado          Boolean   @default(false)
  alineas         String[]    @default([])
  observacoes     String?
  documentos      String[]    @default([])
  // PENDENTE | ASSINADO | HOMOLOGADO | CANCELADO
  status          String      @default("PENDENTE")
  documentoNome   String? // nome do PDF gerado
  documentoGeradoEm DateTime?
  // Cancelamento (art. 5 CLT): o registro NÃO é apagado, entra em CANCELADO
  // e o funcionário volta a ATIVO. O histórico fica preservado.
  canceladoEm      DateTime?
  canceladoPorId   String?
  motivoCancelamento String?
  // Arquivamento: 30 dias após o desligamento o cadastro sai da operação e
  // passa a arquivado. `arquivadoEm` nulo = ainda dentro da janela.
  arquivadoEm      DateTime?
  // Impressão do dossiê completo (advertências + suspensões + rescisão).
  dossieGeradoEm   DateTime?
  origem          Origem      @default(MANUAL)
  createdAt       DateTime    @default(now())
  updatedAt       DateTime    @updatedAt

  @@index([empresaId])
  @@index([empresaId, funcionarioId])
  @@index([empresaId, status])
  // Suporta a varredura de "quem venceu os 30 dias": só os desligamentos ainda
  // dentro da janela entram no índice com `arquivadoEm` nulo.
  @@index([arquivadoEm, status])
}

//  Seção 22 — Folha de Pagamento
//
//  DECISÃO DE PROJETO: por que estes modelos são novos e não reaproveitados
//  --------------------------------------------------------------------------
//  O `Pagamento` que já existe no schema é da ASSINATURA SaaS (vinculado a
//  `Assinatura`, com `assinaturaId` e cobranca de 19,90/mês). Não é folha de
//  pagamento de funcionário — reutilizá-lo misturaria a receita da plataforma com
//  obrigação trabalhista, e a��o tém empresas deēfusos de vida diferentes. Por
//  isso a folha tem modelos próprios, e `PagamentoRH` é o nome deliberado.
//
//  O cálculo NÃO fica no schema: `Folha.lancamentos` guarda o resultado apurado

//  fechada não pode mudar porque o salário foi reajustado depois.

// --- Sincronização automática do Mercado Livre --------------------------------
// Estado-máquina das tarefas recorrentes de sync (produtos / estoque /
// pedidos). POR QUE UMA TABELA PRÓPRIA: o agendamento precisa sobreviver a
// reinicialização, ser recuperável e seguro entre múltiplas instâncias —
// `setInterval` em memória perde o estado no restart e duplica execução com
// mais de um processo. Uma linha por (empresa, tipo) concentra o que falta
// executar e a evidência do que já foi feito (sem duplicar a semântica de
// `Importacao`, que registra cada execução individual).
//
// CONCORRÊNCIA: a trava é o par status=EXECUTANDO + leaseExpiraEm. A execução
// é tomada com updateMany condicional (count=1 => venceu) — operação atômica
// do PostgreSQL, segura entre instâncias sem lock distribuído. Lease vencido
// = execução órfã (processo caiu no meio): o próximo tick desengata e conta
// a recuperação em `recuperadas`.
model SincronizacaoTarefa {
  id                 String    @id @default(cuid())
  empresaId          String
  empresa            Empresa   @relation(fields: [empresaId], references: [id], onDelete: Restrict)
  // Mesmos identificadores dos tipos de Importacao (MERCADO_LIVRE_*): o
  // histórico de execuções e a tarefa falando a mesma língua.
  tipo               String    // MERCADO_LIVRE_PRODUTOS | MERCADO_LIVRE_ESTOQUE | MERCADO_LIVRE_PEDIDOS
  status             String    @default("AGUARDANDO") // AGUARDANDO | EXECUTANDO | PAUSADA
  // Execução vencida = candidata no tick; índice cobre a fila de vencidas.
  proximaExecucaoEm  DateTime
  // Lease da execução em andamento; ao expirar, qualquer instância desengata.
  leaseExpiraEm      DateTime?
  instanciaId        String?
  tentativas         Int       @default(0) // falhas consecutivas (backoff exponencial)
  recuperadas        Int       @default(0) // execuções órfãs desengatadas (lease vencido)
  ultimaExecucaoEm   DateTime?
  ultimoSucessoEm    DateTime?
  // Redigido por construção: guarda apenas { motivo, em } — nunca token,
  // segredo, cabeçalho ou corpo de requisição.
  ultimoErro         Json?
  processadosUltima  Int       @default(0)
  falhasUltima       Int       @default(0)
  createdAt          DateTime  @default(now())
  updatedAt          DateTime  @updatedAt

  @@unique([empresaId, tipo])
  @@index([status, proximaExecucaoEm])
}

// =============================================================================
// Radar Organizacional — camada de inteligência que cruza os módulos
// =============================================================================
//
// POR QUE CADA TABELA EXISTE (nenhuma duplica estrutura existente):
//
// 1) AvaliacaoComportamental — o prompt do Radar diz que o perfil DISC é
//    "informação levantada pelo RH", ou seja, ENTRADA humana, não medição
//    automática. Sem um lugar para o RH registrar, o campo é sempre null e o
//    Radar mostraria um espaço permanentemente vazio. Nenhum modelo existente
//    guarda isso.
//
// 2) PesquisaBemEstar — mesma lógica: bem-estar é perception humana coletada
//    pelo RH. O Radar só faz o cruzamento; quem mede é a pessoa.
//
// 3) RegistroGamificacao — o prompt proíbe localStorage para informação
//    empresarial. Pontos, conquistas e reconhecimento precisam sobreviver a
//    troca de navegador e ser auditáveis.
//
// 4) AcaoRadar — o botão "Criar Plano de Ação" precisa persistir o que o
//    gestor planejou a partir de um alerta. Hoje ele não grava nada.
//
// Ponto comum: as quatro são multi-tenant (empresaId) e ficam fora do modelo
// de decisão — o Radar mostra sinais, não diagnostica.

enum PerfilDisc {
  D
  I
  S
  C
}

// ===========================================================================
//  Ciclo 1 — Plano de Desenvolvimento DISC
// ===========================================================================
//  Transforma uma avaliação DISC REAL em um plano de desenvolvimento
//  individual.
//
//  POR QUE UMA TABELA NOVA E NÃO REAPROVEITAR `AcaoRadar`
//  --------------------------------------------------
//  `AcaoRadar` é o plano de ação derivado de um ALERTA. O plano de
//  desenvolvimento parte de uma AVALIAÇÃO, e a diferença não é de campo, é de
//  domínio: `AcaoRadar.alertaRef` e `alertaTitulo` são NOT NULL porque a linha
//  só existe quando existe um alerta, e é esse `alertaRef` que faz o badge do
//  alerta virar "Em ação"/"Resolvido" (radar.routes.ts). Um plano de
//  desenvolvimento não tem alerta de origem.
//
//  Forçar o reuso exigiria tornar as duas colunas opcionais e acrescentar
//  seis campos (avaliacaoId, objetivo, competencia, descricao, dataInicio,
//  analiseIA) — e a tabela passaria a guardar duas coisas diferentes com o
//  mesmo nome. Criar a entidade própria é o caminho mais simples de explicar e
//  de manter, que é o critério que o próprio schema usa nas outras decisões.
//
//  RELAÇÕES
//  ----------
//  * `funcionario` — de quem é o plano (CASCADE: plano não sobrevive à pessoa).
//  * `avaliacao`   — a avaliação que o originou. Guardar o id, e não só o
//                    perfil, permite responder "com qual avaliação este plano
//                    foi feito?" depois que a vigente mudar.
//
//  SNAPSHOT
//  --------
//  `perfil` e `scores` são cópia do momento da criação. Sem eles, o histórico
//  mudaria de significado sozinho quando uma avaliação nova entrasse: o plano
//  de 2026 passaria a exibir o perfil de 2027.
//
//  `analiseIA` guarda o que a IA sugeriu e o que o gestor aceitou, editou ou
//  recusou. A IA escreve em UM campo só, nunca em `perfil`/`scores`.

enum StatusPlanoDesenvolvimento {
  PLANEJADO
  EM_ANDAMENTO
  CONCLUIDO
  CANCELADO
}

enum ObjetivoDesenvolvimento {

  GESTAO_DE_CONFLITOS
  ATENDIMENTO
  OUTRO

  LIDERANCA
  ORGANIZACAO

  ADAPTABILIDADE
  OUTRO
}

  TREINAMENTO
  ACOMPANHAMENTO
  FEEDBACK
  EXERCICIO_PRATICO
  MENTORIA
  OUTRO
}

// ===========================================================================
//  Ciclo 2 — Treinamentos, participantes e presença
// ===========================================================================
//  POR QUE ENTIDADES NOVAS
//  -----------------------
//  Não existe nada de capacitação no schema: um treinamento é um EVENTO com
//  data, hora, instrutor e local, e nada disso existe em `AvaliacaoComportamental`
//  (que é uma medição pontual), em `PlanoDesenvolvimento` (que é um compromisso
//  de desenvolvimento) ou em `AcaoRadar` (que é reação a um alerta). A ação
//  `TREINAMENTO` do plano de desenvolvimento é um TEXTO dentro de um array — não
//  tem data, não tem sala, não tem lista de quem compareceu. Usar um desses
//  para "agendar" seria guardar um evento como se fosse uma nota.
//
//  RELAÇÃO COM O PLANO DE DESENVOLVIMENTO (Ciclo 1)
//  ------------------------------------------------
//  São entidades diferentes e complementares: o plano diz O QUE desenvolver, o
//  treinamento é UM evento que pode servir a esse plano. Por isso a ligação é
//  explícita e opcional (`planoDesenvolvimentoId`), e não implícita: um
//  treinamento de segurança pode existir sem plano, e um plano pode ter ações
//  que não viram treinamento.
//
//  POR QUE `ParticipanteTreinamento` E NÃO UM ARRAY JSON
//  ---------------------------------------------------
//  Os outros objetos do projeto guardam listas em `Json` (`acoes` do plano,
//  `causas` do alerta) porque são texto livre, consultado nunca. Aqui a lista
//  É a consulta: quantas pessoas tomado presença, quem faltou, o histórico
//  de um funcionário. Isso exige linha, índice e chave única — o que o schema
//  já faz com `@@unique([funcionarioId, data])` no `Ponto`.
//
//  RELAÇÃO COM O PONTO
//  -------------------
//  Deliberadamente SEPARADA. Presença em treinamento não é batida de ponto:
//  não gera falta, não gera hora extra, não entra no espelho de ponto. Um
//  treinamento de 2h não pode aparecer como "2 horas trabalhadas" na folha.
//  Misturar os dois contaminaria o cálculo de jornada com capacitação.
//
//  HORÁRIO
//  --------
//  `horaInicio`/`horaFim` são `String` no formato `HH:MM`, e não `@db.Time`, por
//  exatamente a mesma razão documentada em `Funcionario.horarioEntradaPrevisto`:
//  horário não tem fuso, e um tipo de data/hora no banco convida o bug de
//  fuso que o projeto já cometeu três vezes.

enum StatusTreinamento {
  PLANEJADO
  AGENDADO
  EM_ANDAMENTO
  CONCLUIDO
  CANCELADO
}

enum PrioridadeTreinamento {
  BAIXA
  MEDIA
  ALTA
}

/// `CONVIDADO` e `CONFIRMADO` descrevem a intenção; `PRESENTE`/`AUSENTE`
/// descrevem o que aconteceu. São estados diferentes de propósito: um
/// treinamento CONCLUIDO com gente ainda em `CONFIRMADO` tem ausência não
/// registrada, e a tela precisa poder dizer isso em vez de mostrar 100%
/// presente por omissão.
enum StatusParticipante {
  CONVIDADO
  CONFIRMADO
  PRESENTE
  AUSENTE
  CANCELADO
}

model Treinamento {

  /// Catálogo livre de texto (ex.: "COMUNICACAO", "SEGURANCA"). Sem tabela de
  /// categorias: o projeto usa `categoria String` no `Produto` e `tipo String`
  /// na `Meta`, e uma tabela a mais só traria rigidez sem ganho — o filtro é
  /// por valor, e o valor é legível.
  categoria     String
  objetivo      String
  /// Nome de quem ministra. Texto, e não relação com `Usuario`: o instrutor

/// N:N entre treinamento e funcionário, com o registro de presença.

  @@unique([treinamentoId, funcionarioId])
  // Histórico do funcionário: a ordem de quem treinou quando.
  @@index([empresaId, funcionarioId, dataRegistro])

  @@index([empresaId, treinamentoId, status])

  @@index([empresaId, tipo])
  /// Ciclo 5: quem automatizou este registro, e o quando — é o que separa um
  /// ponto de apuração de um lançamento manual do gestor.
  @@index([empresaId, criterioId])
}

// ===========================================================================
//  Ciclo 4 — Loja de Recompensas e Resgates
// ===========================================================================
//  POR QUE DUAS TABELAS E NÃO UMA
//  --------------------------
//  A `Recompensa` é o CATÁLOGO: o que a empresa oferece, quanto custa, quanto
//  tem. O `Resgate` é a TRANSAÇÃO: quem pegou o quê, quando e em que status.
//  Uma tabela só misturaria "a empresa tem 10 vale-presente" com "a Ana pegou
//  um deles" — e o estoque não se contaria.
//
//  SALDO DE PONTOS — A CONTA CORRETA
//  -------------------------------
//  `RegistroGamificacao` só SOMA pontos. Um resgate precisa DEBITAR, e o
//  saldo não pode ser `soma(pontos)` por três motivos:
//    1. resgate cancelado devolve os pontos;
//    2. registro com status ANULADO precisa deixar de contar;
//    3. saldo é histórico de saldo, não soma de coluna.
//
//  Por isso o saldo é SEMPRE: soma(registros ATIVOS) − soma(resgates que
//  consumiram pontos). É a fórmula que `saldoDePontos` aplica, e ela está
//  testada. Nenhum número de saldo é gravado em coluna — se fosse, deixaria de
//  bater com os registros no primeiro lançamento.
//
//  ESTOQUE
//  -------
//  `quantidadeDisponivel` é o estoque real, e um resgate SOLICITADO já o
//  reserva: é o que impede que 3 pessoas resgatem o último vale-presente antes
//  de alguém aprovar. Cancelar e rejeitar devolvem ao estoque, na mesma
//  transação que muda o status.
//
//  ESTORNO
//  -------
//  Rejeitar/cancelar NÃO apaga a linha: ela vira histórico. E o débito dos
//  pontos é desfeito pela própria fórmula do saldo (o resgate deixa de contar),
//  e não por um segundo lançamento negativo — que duplicaria a conta e
//  esconderia o erro.

enum CategoriaRecompensa {
  BENEFICIO
  EXPERIENCIA
  CURSO
  ALIMENTACAO
  FOLGA
  BRINDE
  OUTRO
}

enum StatusResgate {
  SOLICITADO
  APROVADO
  ENTREGUE
  REJEITADO
  CANCELADO
}

model Recompensa {
  id            String   @id @default(cuid())
  empresaId     String
  nome          String
  descricao     String
  categoria     CategoriaRecompensa @default(OUTRO)
  /// Quanto custa em pontos. `Int`, porque ponto é contagem.
  custoPontos   Int
  /// Estoque REAL. `0` significa "esgotado" e some da vitrine — não
  /// "sem limite", que seria outra decisão e teria outro campo.
  quantidadeDisponivel Int    @default(0)
  /// `false` esconde da loja sem apagar o histórico de quem já resgatou.
  ativa         Boolean  @default(true)
  observacoes   String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  resgates ResgateRecompensa[]

  @@index([empresaId, ativa])
  @@index([empresaId, categoria])
}

model ResgateRecompensa {
  id            String   @id @default(cuid())
  empresaId     String
  recompensaId  String
  recompensa    Recompensa @relation(fields: [recompensaId], references: [id], onDelete: Cascade)
  /// Quem resgatou. Coluna escalar, padrão do projeto; o nome é resolvido em
  /// consulta à parte.
  funcionarioId String
  /// Custo congelado no momento do resgate. Se a empresa reajustar o preço
  /// depois, o histórico continua dizendo o que a pessoa pagou — sem isso, o
  /// registro mentiria sobre a transação antiga.
  pontosUtilizados Int
  status        StatusResgate @default(SOLICITADO)
  /// Quem aprovou/rejeitou/cancelou. NULL enquanto ninguém decidiu.
  aprovadoPorId String?
  data          DateTime @db.Date
  aprovadoEm    DateTime? @db.Date
  entregueEm    DateTime? @db.Date
  observacoes   String?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  @@index([empresaId, funcionarioId, data])
  @@index([empresaId, status])
}

// ===========================================================================
//  Ciclo 5 — Pontuação CONFIGURÁVEL por empresa
// ===========================================================================
//
//  A REGRA DO CICLO, EM UMA FRASE
//  ---------------------------------
//  O NEXUS não tem opinião sobre quanto vale absenceiro. A EMPRESA decide quais
//  critérios existem, qual condição cada um exige, quantos pontos vale e como o
//  saldo se gasta. O NEXUS só faz seis coisas, e nenhuma delas é opinar:
//    1. lê o evento real do funcionário (falta, meta, advertência, treino…);
//    2. compara com a condição que A EMPRESA configurou;
//    3. credita a quantidade de pontos que A EMPRESA configurou;
//    4. guarda o histórico;
//    5. calcula o saldo;
//    6. deixa o saldo ser gasto nas recompensas que A EMPRESA cadastrou.
//
//  POR QUE NÃO HÁ VALOR PADRÃO NESTE ARQUIVO
//  -------------------------------------------
//  Um `pontos: 10` aqui seria uma política do NEXUS valendo para toda empresa
//  que usa o sistema — exatamente o que este ciclo veio desfazer. Por isso os
//  campos de valor e pontos são obrigatórios na criação e a empresa nasce com
//  ZERO critérios: sem configuração, nada é creditado, e a tela diz isso.
//
//  A TABELA É POR EMPRESA
//  ---------------------
//  `empresaId` está em todas as consultas, e os dois índices começam por ele.
//  Duas empresas podem ter exatamente os mesmos nomes de critério com valores
//  completamente diferentes — e isso é o comportamento correto, não um bug.
//
//  `AvaliacaoCriterio` é o que impede a pontuação de inflar: um critério só
//  credita UMA vez por funcionário por período. Rodar a avaliação dez vezes no
//  mesmo mês continua valendo o mesmo ponto, e a evidência do que foi medido
//  fica congelada na linha — se a falta da empresa mudar depois, o histórico
//  continua dizendo o que o NEXUS viu na apuração.

/// Eventos que o NEXUS consegue VERIFICAR no banco. Não são uma lista de
/// metas: são as perguntas que os dados da empresa permitem responder.
enum EventoPontuacao {
  /// Nenhuma falta registrada no período.
  SEM_FALTA
  /// Nenhum atraso somado no período.
  SEM_ATRASO
  /// Nenhuma advertência no período.
  SEM_ADVERTENCIA
  /// Percentual de uma `Meta` do tipo FUNCIONARIO atingiu o parametro.
  META_BATIDA
  /// Quantos treinamentos o funcionário PRESENDEU no período. O enum
  /// `StatusParticipante` tem PRESENTE/AUSENTE, e nao "CONCLUIDO": o evento
  /// mede presenca, que e o dado que existe de fato.
  TREINAMENTO_PRESENTE
  /// Horas extras somadas no período.
  HORAS_EXTRAS
  /// Valor vendido no período.
  VENDAS_NO_PERIODO
}

/// Como a condição do critério é comparada com o valor medido.
enum OperadorCriterio {
  /// medido == valor  (ex.: 0 faltas)
  IGUAL
  /// medido >= valor (ex.: 100% da meta)
  MAIOR_OU_IGUAL
  /// medido <= valor (ex.: no máximo 2 atrasos)
  MENOR_OU_IGUAL
}

/// Janela em que o critério é medido. Define também a chave de idempotência.
enum PeriodoCriterio {
  MENSAL
  TRIMESTRAL
  ANUAL
}

model CriterioPontuacao {
  id            String   @id @default(cuid())
  empresaId     String
  /// Nome que A EMPRESA dá ao seu critério. Texto livre: "Zero falta", "Bateu
  /// a meta do trimestre", "Ajudou a equipe" — o NEXUS não padroniza a política.
  nome          String
  descricao     String?
  evento        EventoPontuacao
  operador      OperadorCriterio @default(IGUAL)
  /// Limiar comparado com o valor medido. Significado conforme o evento:
  /// 0 faltas, 100 (percentual), 2 treinamentos, 10 horas…
  valor         Int      @default(0)
  /// Quantos pontos A EMPRESA paga por este critério. Pode ser NEGATIVO: é
  /// assim que uma empresa pune atraso ou advertência sem inventar um segundo
  /// tipo de registro. Zero é válido e significa "criterio de observacao".
  pontos        Int
  periodo       PeriodoCriterio @default(MENSAL)
  /// `false` desliga o critério sem apagar o histórico já creditado.
  ativo         Boolean  @default(true)
  observacoes   String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  avaliacoes    AvaliacaoCriterio[]
  /// Pontos que este critério já creditou. `SetNull` no outro lado: apagar o
  /// critério não pode apagar o saldo que ele gerou.
  registros    RegistroGamificacao[]

  @@index([empresaId, ativo])
  @@index([empresaId, evento])
}

model AvaliacaoCriterio {
  id            String   @id @default(cuid())
  empresaId     String
  criterioId    String
  criterio      CriterioPontuacao @relation(fields: [criterioId], references: [id], onDelete: Cascade)
  funcionarioId String
  /// O que o NEXUS mediu de fato ("faltas=0", "meta=104%"). É a prova de que
  /// o ponto veio de dado real e não de um chute — e fica congelado.
  evidencia    String
  /// Valor medido no momento da apuração, para a tela mostrar o porquê.
  valorMedido  Int      @default(0)
  /// Pontos creditados, congelados. Se a empresa mudar `pontos` no critério
  /// depois, o histórico continua dizendo quanto foi pago naquele dia.
  pontos       Int
  /// Identificador do período ("2026-09", "2026-Q1", "2026"). Entra na chave
  /// de idempotência, e é o que impede o mesmo mês de pagar duas vezes.
  periodo      String
  /// Registro de gamificação gerado por esta avaliação, se houver.
  registroId   String?
  createdAt    DateTime @default(now())

  /// Um critério credita UMA vez por funcionário por período. Rodar a
  /// avaliação de novo no mesmo mês não soma o mesmo ponto duas vezes.
  @@unique([criterioId, funcionarioId, periodo])
  @@index([empresaId, periodo])
  @@index([empresaId, funcionarioId])
}

model AcaoRadar {
  id            String      @id @default(cuid())
  empresaId     String
  /// Alerta de origem. Guardamos o código/título para o plano não quebrar se

// ===========================================================================
//  Ciclo 3 — Liderança Empática (registro de feedback)
// ===========================================================================
//  POR QUE UMA TABELA NOVA E NÃO REAPROVEITAR `RegistroGamificacao`
//  ---------------------------------------------------------------------
//  `RegistroGamificacao` é PLACAR: guarda pontos, a regra que gerou o ponto e o
//  evento de origem. Ele responde "quantos pontos a pessoa tem".
//
//  Feedback de liderança é CONVERSA: precisa guardar o que foi reconhecido, o
//  que precisa desenvolver, o que ficou acordado e QUANDO será revisado. Nada
//  disso cabe em `descricao String` + `pontos Int` — forçar significaria
//  transformar "o gestor conversou com a pessoa" em "+20 pontos", que é o
//  oposto de liderança empática.
//
//  A mesma lógica já valeu para `PlanoDesenvolvimento` e `Treinamento`: cada
//  entidade nova é porque o DOMÍNIO é outro, não porque a anterior estava
//  "ocupada".
//
//  QUEM É "LÍDER" — DECIDIDO COM VOCÊ
//  ---------------------------------
//  Não existe campo `lideranca` no `Funcionario`, e não foi criado um. A
//  liderança é inferida do CARGO, que já existe e já é do gestor. A lista de
//  termos está em `src/modules/rh/lideranca.ts` (`TERMOS_LIDERANCA`) e é
//  VISÍVEL na tela — se ninguém cadastrar o cargo como "Líder", ninguém é
//  considerado líder, e o sistema diz isso em vez de inventar.
//
//  `proximoAcompanhamento` é DATE e não TIMESTAMP pelo mesmo motivo do resto
//  do schema: é o dia do fato, sem hora e sem fuso.

enum TipoFeedback {
  /// Reconhecimento do que foi bem feito.
  POSITIVO
  /// Conversa sobre o que precisa desenvolver.
  DESENVOLVIMENTO
  /// Alinhamento de expectativas e acordos.
  ACORDO
}

enum StatusAcompanhamento {
  PENDENTE
  REALIZADO
  CANCELADO
}

model FeedbackLideranca {
  id            String   @id @default(cuid())
  empresaId     String
  /// Quem recebeu o feedback.
  funcionarioId String
  funcionario   Funcionario @relation(fields: [funcionarioId], references: [id], onDelete: Cascade)
  /// Quem deu o feedback. Coluna escalar, como `AcaoRadar.responsavelId`: o
  /// aplicador é sempre um `Usuario` desta empresa.
  aplicadoPorId String
  data          DateTime @db.Date
  tipo          TipoFeedback @default(POSITIVO)

  /// Os três blocos que o gestor escreve. `acordos` é o que ficou combinado
  /// — é o campo que diferencia um feedback de um registro de conversa.
  pontosPositivos    String
  pontosADesenvolver String
  acordos            String?

  /// Quando a conversa volta. NULL significa "sem data marcada", que é uma
  /// escolha do gestor — não "esquecido pelo sistema".
  proximoAcompanhamento DateTime? @db.Date
  status         StatusAcompanhamento @default(PENDENTE)

  /// Data em que o acompanhamento marcado aconteceu.
  realizadoEm    DateTime? @db.Date
  observacoes    String?
  origem         Origem    @default(MANUAL)
  createdAt      DateTime  @default(now())
  updatedAt      DateTime  @updatedAt

  @@index([empresaId, data])
  @@index([empresaId, funcionarioId])
  /// A fila de acompanhamento: quem está devendo conversa.
  @@index([empresaId, status, proximoAcompanhamento])
}
